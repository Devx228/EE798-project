\section{Dataset and Experimental Setup}
\label{sec:setup}
% Guideline 2.8: data, sizes, splits, preprocessing, protocol, hardware/software, hyperparameters, baselines.

\paragraph{Tasks (synthetic, generated on the fly).}
\begin{itemize}
  \item \textbf{MQAR} \citep{arora2024zoology}: $n$ key--value pairs then $n$ queries; vocabulary 4096;
        $T\in\{256,512\}$, $n\in\{8,16,32,64\}$.
  \item \textbf{Selective copying} \citep{gu2023mamba}: 16 tokens among noise; train $T=256$, test $T=1024$.
  \item \textbf{Group words}: prefix products in $\mathbb{Z}_2$, $\mathbb{Z}_3$, $S_3$; train $T=64$,
        test $T\in\{256,512\}$.
\end{itemize}
Training data are sampled fresh every step (no fixed training set); evaluation uses fixed held-out
