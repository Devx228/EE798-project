\subsection{Task as a sequence-labelling problem}
A model $f_\theta$ maps a token sequence $x_{1:T}$ to per-position distributions over a vocabulary
$\mathcal{V}$. With a mask $m_t\in\{0,1\}$ marking scored positions, training minimises
\begin{equation}
  \mathcal{L}(\theta) = -\frac{1}{\sum_{n,t} m^{(n)}_t}\sum_{n=1}^{N}\sum_{t=1}^{T} m^{(n)}_t
  \log p_\theta\!\left(y^{(n)}_t \mid x^{(n)}_{1:t}\right).
\end{equation}

\subsection{Selective state-space model}
Continuous-time system $\dot h(\tau) = A h(\tau) + B x(\tau)$, $y = C h$. Zero-order-hold
discretisation with an input-dependent step $\Delta_t$ and scalar $A=-e^{\alpha}<0$ per head gives
\begin{align}
  a_t &= \exp(\Delta_t A) \in (0,1), \qquad \Delta_t = \operatorname{softplus}(w^\top u_t + b),\\
  h_t &= a_t\, h_{t-1} + (\Delta_t x_t)\, B_t^\top, \qquad h_t \in \R^{P\times N},\\
  y_t &= h_t C_t + D x_t .
\end{align}

approximation of the input term used by Mamba-2 and what Mamba-3's trapezoidal rule changes.}

\paragraph{Unrolled (dual) form.} Unrolling the recurrence,
\begin{equation}
  y_t = \sum_{j\le t} \underbrace{\Big(\prod_{k=j+1}^{t} a_k\Big)}_{\text{decay mask } L_{tj}}
        (C_t^\top B_j)\, \Delta_j x_j ,
\end{equation}
i.e.\ causal attention with scores $C_t^\top B_j$ and a multiplicative decay mask \citep{dao2024ssd}.

\subsection{Gated delta rule}
With $\alpha_t\in(0,1]$, $\beta_t\in(0,1)$ (or $(0,2)$) and $\lVert k_t\rVert=1$,
\begin{align}
  S_t &= \alpha_t\,(I - \beta_t k_t k_t^\top)\, S_{t-1} + \beta_t\, k_t v_t^\top, \qquad S_t\in\R^{d_k\times d_v},\\
  o_t &= S_t^\top q_t .
\end{align}

\paragraph{Chunkwise (UT-transform) form.} Let $\gamma_i=\sum_{k\le i}\log\alpha_k$ inside a chunk and
$\Gamma_i = e^{\gamma_i}$. The vectors actually written, $u_i=\beta_i(v_i-\alpha_i S_{i-1}^\top k_i)$,
solve the unit lower-triangular system
\begin{equation}
  (I+M)\,U = \operatorname{diag}(\beta)\,V - \operatorname{diag}(\beta\odot\Gamma)\,K S_0, \qquad
  M_{ij} = \beta_i\, e^{\gamma_i-\gamma_j}\, k_i^\top k_j \ (j<i),
\end{equation}
after which $O = (Q\odot\Gamma)S_0 + (QK^\top\odot \mathcal{D})\,U$ with decay mask
$\mathcal{D}_{ij}=e^{\gamma_i-\gamma_j}\mathbb{1}[j\le i]$. \todo{Derive; complexity.}

\subsection{Beyond negative eigenvalues: rotations and products of reflections}
\label{sec:math-ext}
\paragraph{Rotational SSM (Mamba-3 style).} Group the $N$ state dimensions into $N/2$ pairs and read
each pair as a complex number. The transition becomes a decay \emph{and} an input-dependent rotation,
\begin{equation}
  h_t = a_t\, R(\theta_t)\, h_{t-1} + (\Delta_t x_t) B_t^\top, \qquad
  R(\theta) = \operatorname{diag}\!\Big(\begin{bmatrix}\cos\theta_j & -\sin\theta_j\\ \sin\theta_j & \cos\theta_j\end{bmatrix}\Big)_{j=1}^{N/2},
\end{equation}
with $\theta_t = \pi\tanh(W_\theta u_t)\in(-\pi,\pi)^{N/2}$. Since rotations of the same plane commute,
with $\Phi_t=\sum_{k\le t}\theta_k$ the substitution $\tilde h_t = h_t R(-\Phi_t)^\top$ gives
\begin{equation}
  \tilde h_t = a_t \tilde h_{t-1} + (\Delta_t x_t)\,\tilde B_t^\top,\quad y_t = \tilde h_t \tilde C_t,\qquad
  \tilde B_t = R(-\Phi_t)B_t,\ \tilde C_t = R(-\Phi_t)C_t ,
\end{equation}
so the rotation can be moved onto $B$ and $C$ (a data-dependent RoPE) and the real chunked scan is
reused unchanged. \todo{Derive the substitution. Explain why $\theta=2\pi/3$ implements
$\mathbb{Z}_3$, and why $\theta=\pi$ recovers the negative eigenvalue $-1$.}

\paragraph{DeltaProduct.} Each token applies $n_h$ delta-rule steps with separate keys, values and
write strengths:
\begin{equation}
  S_t = \Big(\prod_{j=1}^{n_h} \big(I - \beta_{t,j} k_{t,j}k_{t,j}^\top\big)\Big)\alpha_t S_{t-1} + \text{(write terms)}.
\end{equation}
With $\beta_{t,j}=2$ each factor is a reflection $H_j$, and $H_2H_1$ is a rotation by twice the angle
between $k_{t,1}$ and $k_{t,2}$ ($\det = +1$). Hence $n_h=2$ can express $\mathbb{Z}_m$ and the
rotations in $S_3$, which one reflection per token cannot. \todo{Prove $\det(H_2H_1)=1$ and the angle
doubling. Implementation: interleave the sub-steps into a sequence of length $n_h T$ (decay at the
first sub-step, read-out after the last), reusing the chunked algorithm.}
